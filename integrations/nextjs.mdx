# Next.js Integration

Complete guide for integrating Veridex with Next.js 13+ (App Router) and Next.js 12 (Pages Router).

## Installation

```bash
npm install @veridex/sdk ethers
```

## App Router (Next.js 13+)

### 1. Create SDK Instance

```typescript
// lib/veridex.ts
import { createSDK } from '@veridex/sdk';

export const sdk = createSDK('base', {
  network: (process.env.NEXT_PUBLIC_NETWORK as 'testnet' | 'mainnet') || 'testnet',
  relayerUrl: process.env.NEXT_PUBLIC_RELAYER_URL,
});
```

### 2. Create Client Provider

```tsx
// app/providers.tsx
'use client';

import { createContext, useContext, useState, useEffect, ReactNode } from 'react';
import { sdk } from '@/lib/veridex';

interface VeridexState {
  isConnected: boolean;
  vaultAddress: string | null;
  isLoading: boolean;
}

interface VeridexContextType extends VeridexState {
  connect: () => Promise<void>;
  disconnect: () => void;
}

const VeridexContext = createContext<VeridexContextType | null>(null);

export function VeridexProvider({ children }: { children: ReactNode }) {
  const [state, setState] = useState<VeridexState>({
    isConnected: false,
    vaultAddress: null,
    isLoading: true,
  });

  useEffect(() => {
    // Check for stored credential on mount
    const init = async () => {
      const stored = sdk.passkey.getStoredCredential();
      if (stored) {
        try {
          await sdk.passkey.login();
          setState({
            isConnected: true,
            vaultAddress: sdk.getVaultAddress(),
            isLoading: false,
          });
          return;
        } catch (e) {
          console.error('Auto-login failed:', e);
        }
      }
      setState((s) => ({ ...s, isLoading: false }));
    };
    init();
  }, []);

  const connect = async () => {
    setState((s) => ({ ...s, isLoading: true }));
    try {
      const existing = sdk.passkey.getStoredCredential();
      if (existing) {
        await sdk.passkey.login();
      } else {
        await sdk.passkey.register('user@example.com', 'My Wallet');
      }
      setState({
        isConnected: true,
        vaultAddress: sdk.getVaultAddress(),
        isLoading: false,
      });
    } catch (error) {
      setState((s) => ({ ...s, isLoading: false }));
      throw error;
    }
  };

  const disconnect = () => {
    sdk.passkey.clearCredential();
    setState({ isConnected: false, vaultAddress: null, isLoading: false });
  };

  return (
    <VeridexContext.Provider value={{ ...state, connect, disconnect }}>
      {children}
    </VeridexContext.Provider>
  );
}

export function useVeridex() {
  const context = useContext(VeridexContext);
  if (!context) throw new Error('useVeridex must be used within VeridexProvider');
  return context;
}
```

### 3. Add to Layout

```tsx
// app/layout.tsx
import { VeridexProvider } from './providers';

export default function RootLayout({
  children,
}: {
  children: React.ReactNode;
}) {
  return (
    <html lang="en">
      <body>
        <VeridexProvider>{children}</VeridexProvider>
      </body>
    </html>
  );
}
```

### 4. Create Components

```tsx
// components/ConnectButton.tsx
'use client';

import { useVeridex } from '@/app/providers';

export function ConnectButton() {
  const { isConnected, vaultAddress, isLoading, connect, disconnect } = useVeridex();

  if (isLoading) {
    return <button disabled>Loading...</button>;
  }

  if (isConnected) {
    return (
      <div className="flex items-center gap-2">
        <code>{vaultAddress?.slice(0, 6)}...{vaultAddress?.slice(-4)}</code>
        <button onClick={disconnect}>Disconnect</button>
      </div>
    );
  }

  return <button onClick={connect}>Connect Passkey</button>;
}
```

### 5. Use in Page

```tsx
// app/page.tsx
import { ConnectButton } from '@/components/ConnectButton';

export default function Home() {
  return (
    <main>
      <h1>My Veridex App</h1>
      <ConnectButton />
    </main>
  );
}
```

## Pages Router (Next.js 12)

### 1. Create SDK and Context

Same as above, but place in `context/VeridexContext.tsx`.

### 2. Add to _app.tsx

```tsx
// pages/_app.tsx
import type { AppProps } from 'next/app';
import { VeridexProvider } from '../context/VeridexContext';

export default function App({ Component, pageProps }: AppProps) {
  return (
    <VeridexProvider>
      <Component {...pageProps} />
    </VeridexProvider>
  );
}
```

### 3. Use in Pages

```tsx
// pages/index.tsx
import { useVeridex } from '../context/VeridexContext';

export default function Home() {
  const { isConnected, connect } = useVeridex();

  return (
    <div>
      {isConnected ? (
        <p>Connected!</p>
      ) : (
        <button onClick={connect}>Connect</button>
      )}
    </div>
  );
}
```

```bash
NEXT_PUBLIC_WORMHOLE_QUERY_API_KEY=your-api-key
```

## Server Components Considerations

<Callout type="warning">
  The Veridex SDK uses browser APIs (WebAuthn, localStorage) and must only be used in Client Components.
</Callout>

Always add `'use client'` directive to components using Veridex:

```tsx
'use client';

import { useVeridex } from '@/app/providers';

export function WalletInfo() {
  const { vaultAddress } = useVeridex();
  return <div>{vaultAddress}</div>;
}
```

## API Routes

For server-side operations (checking balances, etc.):

```typescript
// app/api/balance/[address]/route.ts
import { NextRequest, NextResponse } from 'next/server';

export async function GET(
  request: NextRequest,
  { params }: { params: { address: string } }
) {
  const { address } = params;

  // Call relayer API server-side
  const response = await fetch(
    `${process.env.RELAYER_URL}/api/v1/balances/${address}`
  );
  const balances = await response.json();

  return NextResponse.json(balances);
}
```

## Deployment

### Vercel

```bash
vercel
```

Environment variables are set in Vercel dashboard or via CLI:

```bash
vercel env add NEXT_PUBLIC_NETWORK
vercel env add NEXT_PUBLIC_RELAYER_URL
```

### Self-Hosted

```bash
npm run build
npm start
```

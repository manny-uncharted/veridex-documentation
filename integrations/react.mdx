# React Integration

Complete guide to integrating Veridex with React and Next.js applications.

## Setup

### Install Dependencies

```bash
npm install @veridex/sdk ethers
```

### Create SDK Instance

```typescript
// lib/veridex.ts
import { createSDK } from '@veridex/sdk';

export const sdk = createSDK('base', {
  network: process.env.NEXT_PUBLIC_NETWORK as 'testnet' | 'mainnet' || 'testnet',
  relayerUrl: process.env.NEXT_PUBLIC_RELAYER_URL || 'https://relayer.veridex.network',
});
```

## React Context

### Create Context Provider

```tsx
// context/VeridexContext.tsx
import { createContext, useContext, useState, useEffect, ReactNode } from 'react';
import { sdk } from '../lib/veridex';

interface VeridexContextType {
  isConnected: boolean;
  vaultAddress: string | null;
  isLoading: boolean;
  connect: () => Promise<void>;
  disconnect: () => void;
}

const VeridexContext = createContext<VeridexContextType | null>(null);

export function VeridexProvider({ children }: { children: ReactNode }) {
  const [isConnected, setIsConnected] = useState(false);
  const [vaultAddress, setVaultAddress] = useState<string | null>(null);
  const [isLoading, setIsLoading] = useState(true);

  // Check for existing credential on mount
  useEffect(() => {
    const checkConnection = async () => {
      const stored = sdk.passkey.getStoredCredential();
      if (stored) {
        try {
          await sdk.passkey.login();
          setVaultAddress(sdk.getVaultAddress());
          setIsConnected(true);
        } catch (error) {
          console.error('Auto-login failed:', error);
        }
      }
      setIsLoading(false);
    };
    checkConnection();
  }, []);

  const connect = async () => {
    setIsLoading(true);
    try {
      const existing = sdk.passkey.getStoredCredential();
      if (existing) {
        await sdk.passkey.login();
      } else {
        await sdk.passkey.register('user@example.com', 'My Wallet');
      }
      setVaultAddress(sdk.getVaultAddress());
      setIsConnected(true);
    } finally {
      setIsLoading(false);
    }
  };

  const disconnect = () => {
    sdk.passkey.clearCredential();
    setIsConnected(false);
    setVaultAddress(null);
  };

  return (
    <VeridexContext.Provider
      value={{ isConnected, vaultAddress, isLoading, connect, disconnect }}
    >
      {children}
    </VeridexContext.Provider>
  );
}

export function useVeridex() {
  const context = useContext(VeridexContext);
  if (!context) {
    throw new Error('useVeridex must be used within VeridexProvider');
  }
  return context;
}
```

### Use in App

```tsx
// pages/_app.tsx (Next.js) or App.tsx (Create React App)
import { VeridexProvider } from '../context/VeridexContext';

export default function App({ Component, pageProps }) {
  return (
    <VeridexProvider>
      <Component {...pageProps} />
    </VeridexProvider>
  );
}
```

## Components

### Connect Button

```tsx
// components/ConnectButton.tsx
import { useVeridex } from '../context/VeridexContext';

export function ConnectButton() {
  const { isConnected, vaultAddress, isLoading, connect, disconnect } = useVeridex();

  if (isLoading) {
    return (
      <button disabled className="bg-gray-300 text-gray-500 px-4 py-2 rounded-lg">
        Loading...
      </button>
    );
  }

  if (isConnected) {
    return (
      <div className="flex items-center gap-3">
        <span className="font-mono text-sm">
          {vaultAddress?.slice(0, 6)}...{vaultAddress?.slice(-4)}
        </span>
        <button
          onClick={disconnect}
          className="text-red-500 text-sm hover:underline"
        >
          Disconnect
        </button>
      </div>
    );
  }

  return (
    <button
      onClick={connect}
      className="bg-blue-600 text-white px-4 py-2 rounded-lg hover:bg-blue-700"
    >
      Connect with Passkey
    </button>
  );
}
```

### Balance Display

```tsx
// components/BalanceDisplay.tsx
import { useState, useEffect } from 'react';
import { sdk } from '../lib/veridex';
import { useVeridex } from '../context/VeridexContext';

export function BalanceDisplay() {
  const { isConnected } = useVeridex();
  const [balances, setBalances] = useState<Record<string, Record<string, string>>>({});
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    if (isConnected) {
      loadBalances();
    }
  }, [isConnected]);

  const loadBalances = async () => {
    setLoading(true);
    try {
      const result = await sdk.getBalances();
      setBalances(result);
    } catch (error) {
      console.error('Failed to load balances:', error);
    }
    setLoading(false);
  };

  if (!isConnected) return null;

  if (loading) {
    return <div className="animate-pulse">Loading balances...</div>;
  }

  return (
    <div className="space-y-4">
      <div className="flex justify-between items-center">
        <h3 className="font-bold">Balances</h3>
        <button onClick={loadBalances} className="text-sm text-blue-600">
          Refresh
        </button>
      </div>

      {Object.entries(balances).map(([chain, tokens]) => (
        <div key={chain} className="p-3 bg-gray-50 rounded-lg">
          <h4 className="font-medium text-sm text-gray-600 mb-2">{chain}</h4>
          {Object.entries(tokens).map(([symbol, balance]) => (
            <div key={symbol} className="flex justify-between">
              <span>{symbol}</span>
              <span className="font-mono">{balance}</span>
            </div>
          ))}
        </div>
      ))}
    </div>
  );
}
```

### Transfer Form

```tsx
// components/TransferForm.tsx
import { useState } from 'react';
import { parseUnits } from 'ethers';
import { sdk } from '../lib/veridex';
import { useVeridex } from '../context/VeridexContext';

const USDC = '0x036CbD53842c5426634e7929541eC2318f3dCF7e';

export function TransferForm() {
  const { isConnected } = useVeridex();
  const [recipient, setRecipient] = useState('');
  const [amount, setAmount] = useState('');
  const [loading, setLoading] = useState(false);
  const [result, setResult] = useState<{ success: boolean; message: string } | null>(null);

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);
    setResult(null);

    try {
      const tx = await sdk.transferViaRelayer({
        token: USDC,
        recipient,
        amount: parseUnits(amount, 6),
      });

      setResult({
        success: true,
        message: `Transfer successful! Tx: ${tx.txHash}`,
      });
      setRecipient('');
      setAmount('');
    } catch (error) {
      setResult({
        success: false,
        message: error instanceof Error ? error.message : 'Transfer failed',
      });
    }

    setLoading(false);
  };

  if (!isConnected) {
    return (
      <div className="p-4 bg-gray-100 rounded-lg text-center">
        Connect your wallet to send tokens
      </div>
    );
  }

  return (
    <form onSubmit={handleSubmit} className="space-y-4">
      <div>
        <label className="block text-sm font-medium mb-1">Recipient</label>
        <input
          type="text"
          value={recipient}
          onChange={(e) => setRecipient(e.target.value)}
          placeholder="0x..."
          className="w-full p-2 border rounded-lg"
          required
        />
      </div>

      <div>
        <label className="block text-sm font-medium mb-1">Amount (USDC)</label>
        <input
          type="number"
          value={amount}
          onChange={(e) => setAmount(e.target.value)}
          placeholder="10.00"
          step="0.01"
          min="0"
          className="w-full p-2 border rounded-lg"
          required
        />
      </div>

      <button
        type="submit"
        disabled={loading}
        className="w-full bg-blue-600 text-white py-2 rounded-lg disabled:bg-gray-400"
      >
        {loading ? 'Sending...' : 'Send USDC'}
      </button>

      {result && (
        <div
          className={`p-3 rounded-lg ${
            result.success ? 'bg-green-100 text-green-800' : 'bg-red-100 text-red-800'
          }`}
        >
          {result.message}
        </div>
      )}
    </form>
  );
}
```

## Hooks

### useBalances Hook

```tsx
// hooks/useBalances.ts
import { useState, useEffect, useCallback } from 'react';
import { sdk } from '../lib/veridex';

export function useBalances() {
  const [balances, setBalances] = useState<Record<string, Record<string, string>>>({});
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<Error | null>(null);

  const refresh = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      const result = await sdk.getBalances();
      setBalances(result);
    } catch (err) {
      setError(err instanceof Error ? err : new Error('Failed to load balances'));
    }
    setLoading(false);
  }, []);

  useEffect(() => {
    refresh();
  }, [refresh]);

  return { balances, loading, error, refresh };
}
```

### useTransfer Hook

```tsx
// hooks/useTransfer.ts
import { useState, useCallback } from 'react';
import { sdk } from '../lib/veridex';

interface TransferParams {
  token: string;
  recipient: string;
  amount: bigint;
}

export function useTransfer() {
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<Error | null>(null);
  const [txHash, setTxHash] = useState<string | null>(null);

  const transfer = useCallback(async (params: TransferParams) => {
    setLoading(true);
    setError(null);
    setTxHash(null);

    try {
      const result = await sdk.transferViaRelayer(params);
      setTxHash(result.txHash);
      return result;
    } catch (err) {
      const error = err instanceof Error ? err : new Error('Transfer failed');
      setError(error);
      throw error;
    } finally {
      setLoading(false);
    }
  }, []);

  return { transfer, loading, error, txHash };
}
```

### useSession Hook

```tsx
// hooks/useSession.ts
import { useState, useEffect, useCallback } from 'react';
import { sdk } from '../lib/veridex';

export function useSession() {
  const [session, setSession] = useState(sdk.sessions.getActive());
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    // Check session on mount
    setSession(sdk.sessions.getActive());
  }, []);

  const create = useCallback(async (params: { duration: number; maxValue?: bigint }) => {
    setLoading(true);
    try {
      const newSession = await sdk.sessions.create(params);
      setSession(newSession);
      return newSession;
    } finally {
      setLoading(false);
    }
  }, []);

  const revoke = useCallback(async () => {
    await sdk.sessions.revoke();
    setSession(null);
  }, []);

  return { session, create, revoke, loading };
}
```

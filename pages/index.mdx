```typescript
import { createSDK } from '@veridex/sdk';
import { parseEther } from 'ethers';

const sdk = createSDK('base', { network: 'testnet' });                       // Base Sepolia + public relayer
await sdk.passkey.register('alice@example.com', 'My Wallet');                // FaceID / TouchID prompt → vault provisioned
const tx = await sdk.transferViaRelayer({
  targetChain: 10004,                                                        // Base Sepolia (Wormhole chain id)
  token: 'native',
  recipient: '0x742d35Cc6634C0532925a3b844Bc9e7595f5A234',
  amount: parseEther('0.001'),
});
console.log(tx.transactionHash);                                             // on-chain Hub tx hash
```

> Want a key with hard spending caps that signs many transfers without re-prompting? Add a [`SessionManager`](/sdks/sessions) on top of this flow — same SDK, two extra setup lines.
